

   # 快速启动

  # 1. 进入后端项目目录
  cd petlis-backend

  # 2. 配置数据库
  # 编辑 src/main/resources/application-dev.yml

  # 清理并编译项目（会下载依赖）
  mvn clean compile


  # 如果编译成功，运行测试
  mvn test

  # 打包项目
  mvn package -DskipTests

  # 3. 编译项目
  mvn clean package

  # 4. 运行应用
  java -jar target/petlis-backend-1.0.0.jar

  # 5. 访问Swagger文档
  # http://localhost:8080/api/swagger-ui.html

  Docker部署

  # 使用docker-compose一键启动
  cd /Users/gong/claudelife/PETLIS
  docker-compose up -d
