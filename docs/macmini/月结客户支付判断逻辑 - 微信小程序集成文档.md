  一、业务逻辑说明

  1.1 核心业务规则

  月结客户是指按月结算账单的医院客户（PayType = 1），与次结客户（即时支付）相对。

  关键判断点：
  1. ✅ 判断医院是否为月结客户（PayType == 1）
  2. ✅ 根据当前日期计算需要结清的账单月份
  3. ✅ 查询该月份是否有未支付的检测项
  4. ✅ 有未支付账单 → 阻止查看PDF，显示账单
  5. ✅ 无未支付账单 → 允许查看PDF报告

  1.2 账单周期计算规则

  每月20日是账单周期的分界点

  当前日期 >= 20日：需结清【上上个月】的账单
    示例：5月20日 → 需结清3月账单

  当前日期 < 20日：需结清【上个月】的账单
    示例：5月10日 → 需结清4月账单

  1.3 未支付账单查询条件

  查询必须同时满足以下4个条件：

  | 字段            | 条件         | 说明                   |
  |-----------------|--------------|------------------------|
  | InfoHospitalId  | = 医院ID     | 指定医院的记录         |
  | IsCheckPass     | = 0          | 检查通过的有效检测项   |
  | ReceiveSampleDt | 在账单周期内 | 样本接收日期在指定月份 |
  | PayStatus2      | = 0          | 未支付状态             |

  3.2 Java实现代码示例

  package com.laboklin.controller;

  import com.laboklin.entity.DiagnosisDetail;
  import com.laboklin.entity.Hospital;
  import com.laboklin.service.MonthlyPaymentService;
  import com.laboklin.dto.PaymentCheckResponse;
  import org.springframework.beans.factory.annotation.Autowired;
  import org.springframework.web.bind.annotation.*;
  import java.time.LocalDateTime;
  import java.time.format.DateTimeFormatter;
  import java.util.List;

  @RestController
  @RequestMapping("/api/monthly")
  public class MonthlyPaymentController {

      @Autowired
      private MonthlyPaymentService monthlyPaymentService;

      /**
       * 检查月结客户支付状态
       */
      @GetMapping("/payment/check")
      public ApiResponse<PaymentCheckResponse> checkPaymentStatus(
              @RequestParam Integer hospitalId,
              @RequestParam(required = false) Integer diagnosisDetailId) {

          // 1. 查询医院信息
          Hospital hospital = monthlyPaymentService.getHospitalById(hospitalId);
          if (hospital == null) {
              return ApiResponse.error("医院不存在");
          }

          // 2. 判断是否为月结客户
          if (hospital.getPayType() != 1) {
              // 次结客户，直接返回
              return ApiResponse.success(PaymentCheckResponse.notMonthlyCustomer());
          }

          // 3. 计算账单周期（核心逻辑）
          String billPeriod = calculateBillPeriod();

          // 4. 查询未支付账单
          List<DiagnosisDetail> unpaidBills = monthlyPaymentService.getUnpaidBills(
              hospitalId,
              billPeriod
          );

          // 5. 构建响应
          PaymentCheckResponse response = PaymentCheckResponse.builder()
              .isMonthlyCustomer(true)
              .billPeriod(billPeriod)
              .unpaidCount(unpaidBills.size())
              .unpaidAmount(calculateTotalAmount(unpaidBills))
              .build();

          // 6. 判断是否可以查看PDF
          if (unpaidBills.isEmpty()) {
              // 无未支付账单，可以查看PDF
              response.setCanViewPdf(true);
              if (diagnosisDetailId != null) {
                  DiagnosisDetail detail = monthlyPaymentService.getDiagnosisDetail(diagnosisDetailId);
                  response.setPdfUrl(detail != null ? detail.getPdfHttpUrl() : null);
              }
          } else {
              // 有未支付账单，不能查看PDF
              response.setCanViewPdf(false);
              response.setUnpaidBills(unpaidBills);
          }

          return ApiResponse.success(response);
      }

      /**
       * 计算账单周期（核心算法）
       * 
       * 规则：
       * - 每月20日及以后：需结清上上个月的账单
       * - 每月20日以前：需结清上个月的账单
       * 
       * @return 账单月份，格式：YYYY-MM
       */
      private String calculateBillPeriod() {
          LocalDateTime now = LocalDateTime.now();
          int day = now.getDayOfMonth();

          if (day >= 20) {
              // 20日及以后，返回上上个月
              return now.minusMonths(2).format(DateTimeFormatter.ofPattern("yyyy-MM"));
          } else {
              // 20日以前，返回上个月
              return now.minusMonths(1).format(DateTimeFormatter.ofPattern("yyyy-MM"));
          }
      }
  }

  3.3 Service层实现

  package com.laboklin.service.impl;

  import com.laboklin.entity.DiagnosisDetail;
  import com.laboklin.mapper.DiagnosisDetailMapper;
  import com.laboklin.service.MonthlyPaymentService;
  import org.springframework.beans.factory.annotation.Autowired;
  import org.springframework.stereotype.Service;
  import java.time.LocalDateTime;
  import java.util.List;

  @Service
  public class MonthlyPaymentServiceImpl implements MonthlyPaymentService {

      @Autowired
      private DiagnosisDetailMapper diagnosisDetailMapper;

      /**
       * 查询未支付账单
       * 
       * 查询条件：
       * 1. InfoHospitalId = ?
       * 2. IsCheckPass = 0（检查通过）
       * 3. ReceiveSampleDt 在账单周期内
       * 4. PayStatus2 = 0（未支付）
       */
      @Override
      public List<DiagnosisDetail> getUnpaidBills(Integer hospitalId, String billPeriod) {
          // 计算账单周期的起止日期
          String startDate = billPeriod + "-01 00:00:00";
          String endDate = LocalDateTime.parse(startDate)
              .plusMonths(1)
              .minusDays(1)
              .format(DateTimeFormatter.ofPattern("yyyy-MM-dd 23:59:59"));

          // 使用MyBatis查询
          return diagnosisDetailMapper.selectUnpaidBills(hospitalId, startDate, endDate);
      }
  }

  // pages/view-report/view-report.js
  Page({
    data: {
      diagnosisDetailId: null,

    onLoad(options) {

    },

    /**
     * 检查支付状态（核心方法）
     */
    async checkPaymentStatus() {
      wx.showLoading({ title: '加载中...' });

      try {
        const res = await wx.request({
          url: 'https://api.laboklin.cn/api/monthly/payment/check',
          method: 'GET',
          data: {
            hospitalId: this.data.hospitalId,
            diagnosisDetailId: this.data.diagnosisDetailId
          }
        });

        const { code, data } = res.data;

        if (code === 200) {
          if (data.isMonthlyCustomer) {
            // 月结客户逻辑
            this.handleMonthlyCustomer(data);
          } else {
            // 次结客户逻辑
            this.handleNormalCustomer(data);
          }
        } else {
          wx.showToast({ title: '加载失败', icon: 'none' });
        }
      } catch (error) {
        console.error('检查支付状态失败:', error);
        wx.showToast({ title: '网络错误', icon: 'none' });
      } finally {
        wx.hideLoading();
        this.setData({ loading: false });
      }
    },

    /**
     * 处理月结客户逻辑
     */
    handleMonthlyCustomer(data) {
      const { canViewPdf, unpaidBills, billPeriod, unpaidAmount, pdfUrl } = data;

      if (!canViewPdf) {
        // 有未支付账单，显示账单页面
        wx.showModal({
          title: '月结账户提示',
          content: `您是月结账户，${billPeriod}月有未支付账单 ¥${unpaidAmount}，请结清后查看报告`,
          confirmText: '查看账单',
          success(res) {
            if (res.confirm) {
              // 跳转到账单详情页
              wx.navigateTo({
                url: `/pages/bill-detail/bill-detail?hospitalId=${this.data.hospitalId}&month=${billPeriod}`
              });
            }
          }
        });
      } else {
        // 无未支付账单，可以直接查看PDF
        this.viewPdfReport(pdfUrl);
      }
    },

    /**
     * 处理次结客户逻辑
     */
    handleNormalCustomer(data) {
      // 次结客户的支付逻辑
      if (data.needPay) {
        wx.showModal({
          title: '待支付',
          content: `您有账单待支付 ¥${data.amount}，支付后即可查看报告`,
          confirmText: '去支付',
          success(res) {
            if (res.confirm) {
              // 跳转到支付页面
              wx.navigateTo({
                url: `/pages/payment/payment?amount=${data.amount}&orderNo=${data.orderNo}`
              });
            }
          }
        });
      } else {
        this.viewPdfReport(data.pdfUrl);
      }
    },

    /**
     * 查看PDF报告
     */

  七、常见问题FAQ

  Q1: 为什么是每月20日作为分界点？
  A: 这是业务规则，给医院5天时间处理上月账单，20日后的新账单计入下一周期。

  Q2: PayStatus2 的各个值含义？
  A:
  - 0 = 未结（红色显示）
  - 1 = 现金支付
  - 2 = 转账支付
  - 3 = 网银已支付
  - 4 = 忽略（特殊情况标记）

  Q3: IsCheckPass 为什么要是 0？
  A: 0表示检查通过的有效检测项，只有通过检查的才需要支付。

  Q4: 次结客户和月结客户可以切换吗？
  A: 可以，修改 lb_infohospital.PayType 字段即可，建议月初切换。
