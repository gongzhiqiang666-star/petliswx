## 🔴 高优先级问题

### 1. setData 数据冗余问题 (严重)

**位置**: `pages/orders/orders.js:187-199`

**问题代码**:
```javascript
filterOrders() {
  const { activeTab, orders } = this.data

  const filtered = activeTab === 'all'
    ? orders
    : orders.filter(order => String(order.sampleStatus) === activeTab)

  // ❌ 问题：每次筛选都创建新对象并 setData
  const filteredWithLabels = filtered.map(order => ({
    ...order,  // 展开运算符复制整个对象
    statusLabel: this.getStatusLabel(order.sampleStatus),
    collectDt: this.formatDate(order.collectDt)
  }))

  this.setData({ filteredOrders: filteredWithLabels })
}
```

**违反的最佳实践**:
- ❌ 避免 setData 数据冗余
- ❌ 避免不必要的数据复制

**性能影响**:
- 每次筛选都复制整个 orders 数组
- 使用展开运算符 `...order` 复制所有订单属性
- 频繁的 setData 会触发视图层重新渲染

**优化建议**:
```javascript
// ✅ 优化方案1：使用缓存避免重复计算
filterOrders() {
  const { activeTab, orders } = this.data

  // 只在tab切换时重新筛选
  if (this._lastActiveTab === activeTab && this._lastOrdersLength === orders.length) {
    return
  }

  this._lastActiveTab = activeTab
  this._lastOrdersLength = orders.length

  const filtered = activeTab === 'all'
    ? orders
    : orders.filter(order => String(order.sampleStatus) === activeTab)

  // ✅ 只更新必要的数据
  this.setData({
    filteredOrders: filtered,
    // 使用计算属性或wxs处理标签，不在JS中处理
  })
}

// ✅ 优化方案2：在WXML中使用WXS处理状态标签
// orders.wxml
<wxs module="statusHelper">
  module.getStatusLabel = function(status) {
    const labels = {
      '-3': '待支付',
      '-2': '草稿',
      '0': '待审核',
      '1': '审核未通过',
      '2': '等待报告',
      '3': '已完成'
    }
    return labels[String(status)] || '未知'
  }
</wxs>

<view wx:for="{{filteredOrders}}" wx:key="orderId">
  <text>{{statusHelper.getStatusLabel(item.sampleStatus)}}</text>
</view>
```

### 2. onShow 重复加载数据 (严重)

**位置**: `pages/orders/orders.js:34-37`

**问题代码**:
```javascript
onShow() {
  // ❌ 问题：每次显示页面都重新加载订单
  this.resetAndLoadOrders()
}
```

**违反的最佳实践**:
- ❌ 启动性能优化 - 避免不必要的数据请求
- ❌ 缺少缓存策略

**性能影响**:
- 用户从详情页返回时会重新请求订单列表
- 频繁的网络请求增加服务器负担
- 用户体验差，每次返回都看到加载动画

**优化建议**:
```javascript
onShow() {
  // ✅ 优化方案：使用时间戳判断是否需要刷新
  const lastLoadTime = this._lastLoadTime || 0
  const now = Date.now()
  const REFRESH_INTERVAL = 5 * 60 * 1000 // 5分钟

  // 如果距离上次加载超过5分钟，或者列表为空，才重新加载
  if (now - lastLoadTime > REFRESH_INTERVAL || this.data.orders.length === 0) {
    this.resetAndLoadOrders()
    this._lastLoadTime = now
  }

  // ✅ 或者只在首次加载时获取数据
  // this.loadOrdersIfNeeded()
}

// ✅ 更好的方案：使用页面生命周期和全局数据
onLoad() {
  this.loadOrders()
  this._lastLoadTime = Date.now()
}

onShow() {
  // 不自动刷新，用户可下拉刷新
}

onPullDownRefresh() {
  this.resetAndLoadOrders()
  wx.stopPullDownRefresh()
}
```

### 3. 缺少数据缓存策略 (严重)

**位置**: `pages/order-preview/order-preview.js:12-24`

**问题代码**:
```javascript
onLoad() {
  try {
    // ❌ 问题：每次都从 storage 读取，没有缓存
    const formData = wx.getStorageSync('order_form');

    if (!formData || !formData.petInfo) {
      // ...
    }
  } catch (error) {
    // ...
  }
}
```

**违反的最佳实践**:
- ❌ 数据缓存策略缺失
- ❌ 未利用 wx.storage 的缓存机制

**性能影响**:
- 频繁的同步 storage 读取会阻塞主线程
- 重复读取相同数据
- 无缓存失效策略

**优化建议**:
```javascript
// ✅ 优化方案：使用App.globalData缓存
// app.js
App({
  globalData: {
    cache: {
      orderForm: null,
      orderFormTimestamp: 0,
      ORDER_FORM_CACHE_TTL: 30 * 60 * 1000 // 30分钟
    }
  },

  // 获取缓存的订单表单
  getCachedOrderForm() {
    const { orderForm, orderFormTimestamp, ORDER_FORM_CACHE_TTL } = this.globalData.cache
    const now = Date.now()

    if (orderForm && (now - orderFormTimestamp < ORDER_FORM_CACHE_TTL)) {
      console.log('[App] 使用缓存的订单表单')
      return orderForm
    }

    return null
  },

  // 设置订单表单缓存
  setCachedOrderForm(formData) {
    this.globalData.cache.orderForm = formData
    this.globalData.cache.orderFormTimestamp = Date.now()
  }
})

// order-preview.js
const app = getApp()

Page({
  onLoad() {
    // ✅ 优先使用全局缓存
    let formData = app.getCachedOrderForm()

    if (!formData) {
      // 缓存未命中，从storage读取
      try {
        formData = wx.getStorageSync('order_form')
        if (formData) {
          app.setCachedOrderForm(formData)
        }
      } catch (error) {
        console.error('[OrderPreview] 读取storage失败:', error)
      }
    }

    if (!formData || !formData.petInfo) {
      wx.showToast({
        title: '数据错误，请重新填写',
        icon: 'none'
      })
      setTimeout(() => {
        wx.navigateBack()
      }, 1500)
      return
    }

    // ... 处理 formData
  }
})
```

### 6. 日期格式化性能问题 (轻微)

**位置**: `pages/order-preview/order-preview.js:99-125`

**问题代码**:
```javascript
formatDate(dateStr) {
  // ❌ 每次都进行复杂的类型判断和转换
  if (!dateStr) return '';

  let date;

  if (typeof dateStr === 'number') {
    date = new Date(dateStr);
  } else if (typeof dateStr === 'string' && /^\d+$/.test(dateStr)) {
    date = new Date(parseInt(dateStr));
  } else {
    const isoStr = dateStr.replace(' ', 'T');
    date = new Date(isoStr);
  }

  if (isNaN(date.getTime())) return '';

  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}
```

**性能影响**:
- 每个订单都调用一次此函数
- 大量的正则表达式和字符串操作
- 在 filterOrders 中被频繁调用

**优化建议**:
```javascript
// ✅ 优化方案1：缓存格式化结果
formatDate(dateStr) {
  if (!dateStr) return '';

  // 使用缓存
  if (this._dateCache && this._dateCache[dateStr]) {
    return this._dateCache[dateStr];
  }

  // 格式化逻辑...
  const formatted = `${year}-${month}-${day}`;

  // 缓存结果
  if (!this._dateCache) {
    this._dateCache = {};
  }
  this._dateCache[dateStr] = formatted;

  return formatted;
}

// ✅ 优化方案2：使用WXS在视图层处理
<wxs module="dateHelper">
  module.formatDate = function(timestamp) {
    if (!timestamp) return '';

    var date = new Date(timestamp);
    var year = date.getFullYear();
    var month = date.getMonth() + 1;
    var day = date.getDate();

    return year + '-' +
           (month < 10 ? '0' : '') + month + '-' +
           (day < 10 ? '0' : '') + day;
  }
</wxs>

<view>{{dateHelper.formatDate(item.collectDt)}}</view>
```

### 8. 网络请求优化建议 (轻微)

**当前实现**: 已有基本优化
- ✅ 使用了 api 模块统一管理请求
- ✅ 有基本的错误处理

**优化建议**:
```javascript
// ✅ 添加请求队列和并发控制
class RequestQueue {
  constructor(maxConcurrent = 3) {
    this.queue = []
    this.running = 0
    this.maxConcurrent = maxConcurrent
  }

  add(requestFn) {
    return new Promise((resolve, reject) => {
      this.queue.push({ requestFn, resolve, reject })
      this.runNext()
    })
  }

  runNext() {
    if (this.running >= this.maxConcurrent || this.queue.length === 0) {
      return
    }

    this.running++
    const { requestFn, resolve, reject } = this.queue.shift()

    requestFn()
      .then(resolve)
      .catch(reject)
      .finally(() => {
        this.running--
        this.runNext()
      })
  }
}

// 使用
const queue = new RequestQueue()
queue.add(() => api.orders.getList(params))
```
